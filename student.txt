#include <iostream>
#include <string>
using namespace std;

class Student {
private:
    int rollNo;
    string name;
    int age;
    string course;

public:
    Student(int r, string n, int a, string c) {
        rollNo = r;
        name = n;
        age = a;
        course = c;
    }

    void displayProfile() {
        cout << "----- Student Profile -----" << endl;
        cout << "Roll Number : " << rollNo << endl;
        cout << "Name        : " << name << endl;
        cout << "Age         : " << age << endl;
        cout << "Course      : " << course << endl;
    }
};

int main() {
    int rollNo, age;
    string name, course;

    cout << "Enter Roll Number: ";
    cin >> rollNo;
    cin.ignore(); 

    cout << "Enter Name: ";
    getline(cin, name);

    cout << "Enter Age: ";
    cin >> age;
    cin.ignore();

    cout << "Enter Course: ";
    getline(cin, course);

    Student s1(rollNo, name, age, course);

    s1.displayProfile();

    return 0;
}
